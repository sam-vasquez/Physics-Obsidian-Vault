\begin{thm*}{Conformal Killing Vectors in $d \geq 3$}{} 
    The conformal Killing equation in $d\geq 3$ dimensions has four kinds of solutions:
    \begin{align*}
        \epsilon^\mu (x) &= a^\mu \\
        \epsilon^\mu (x) &= m^\mu{}_\nu x^\nu \\
        \epsilon^\mu (x) &= \alpha x^\mu \\
        \epsilon^\mu (x) &= 2(x\cdot b)x^\mu - b^\mu x^2
    \end{align*}
\end{thm*}
